"""
Module 035
Hurst Exponent Estimation

Pipeline: log returns → R/S analysis → statistics → validation → report → export

FIXED: Computes log returns before Hurst analysis. Stores series_type.

Full-sample and rolling Hurst share the same gap-aware log-return
series, so both statistics are computed on identical observations.
"""

from typing import Optional, Dict, Any

import numpy as np
import pandas as pd

from atlas.core.base_research_module import BaseResearchModule
from atlas.core.rolling_market_features import (
    compute_rolling_hurst,
    compute_gap_aware_log_returns,
)
from atlas.framework.registry import AtlasRegistry

from .engine import HurstEngine
from .statistics import HurstStatisticsEngine
from .validator import run_validation
from .report import HurstReport


class HurstResearch(BaseResearchModule):

    MODULE_OUTPUT = "module035"
    METADATA = HurstStatisticsEngine.METADATA

    def __init__(self, registry: Optional[AtlasRegistry] = None):

        super().__init__(
            module_name="Hurst Exponent",
            module_version=self.METADATA["version"],
        )

        self.registry = registry or AtlasRegistry()
        self.engine = HurstEngine(min_window=16, num_windows=20)
        self.stats_engine = HurstStatisticsEngine()
        self.result_hurst = None
        self.statistics = None
        self.rolling_hurst = None

    def _register_pipeline_steps(self):
        self.pipeline_steps = [
            ("hurst", self._compute_hurst),
            ("statistics", self._compute_statistics),
            ("validate", self._validate),
            ("export", self._export_results),
        ]

    def _compute_hurst(self) -> Dict[str, Any]:
        self.logger.info(
            "Step 1: Computing full-sample and rolling Hurst exponent"
        )

        prices = pd.to_numeric(
            self.df["close"],
            errors="coerce",
        ).to_numpy(dtype=float)

        if not np.isfinite(prices).all() or np.any(prices <= 0):
            raise ValueError("Invalid close prices for Hurst calculation.")

        # ----------------------------------------------------------
        # Full-sample return series.
        #
        # Uses the same gap-aware computation as the rolling helper,
        # so both full-sample and rolling Hurst are computed on
        # identically-cleaned log returns: NaNs are present where
        # the timestamp gap is not exactly one expected candle
        # interval, and those rows are dropped before scoring.
        #
        # Requires market_df to be loaded (BaseResearchModule always
        # provides it by the time this step runs — the earlier patch
        # already enforced this for the rolling feature).
        # ----------------------------------------------------------
        if self.market_df is None:
            raise RuntimeError("Module 035 market_df is unavailable.")

        returns = compute_gap_aware_log_returns(
            timestamps=self.market_df["timestamp"],
            close=self.market_df["close"],
        )
        returns = returns[np.isfinite(returns)]

        if len(returns) < 100:
            raise ValueError(
                f"Insufficient valid log returns: {len(returns)}"
            )

        self.result_hurst = self.engine.compute(
            returns,
            series_type="log_returns",
        )

        # New causal rolling feature.
        self.rolling_hurst = compute_rolling_hurst(
            timestamps=self.market_df["timestamp"],
            close=self.market_df["close"],
            window_days=60,
        )

        finite_rolling = self.rolling_hurst[
            np.isfinite(self.rolling_hurst)
        ]

        if len(finite_rolling) < 200:
            raise ValueError(
                "Insufficient finite rolling Hurst observations: "
                f"{len(finite_rolling)} < 200"
            )

        if np.unique(np.round(finite_rolling, 10)).size < 2:
            raise ValueError("Rolling Hurst feature is constant.")

        self.logger.metric("Series type", self.result_hurst.series_type)
        self.logger.metric("Hurst exponent", f"{self.result_hurst.hurst_exponent:.4f}")
        self.logger.metric("Sample size", f"{self.result_hurst.sample_size:,}")
        self.logger.metric("Windows", f"{len(self.result_hurst.scales)}")
        self.logger.metric(
            "Rolling Hurst valid rows",
            f"{len(finite_rolling):,}",
        )
        self.logger.metric(
            "Rolling Hurst mean",
            f"{float(np.mean(finite_rolling)):.4f}",
        )
        self.logger.success("Hurst exponent and rolling feature computed")

        return {
            "hurst_exponent": self.result_hurst.hurst_exponent,
            "sample_size": self.result_hurst.sample_size,
            "series_type": self.result_hurst.series_type,
            "rolling_hurst_valid_rows": len(finite_rolling),
        }

    def _compute_statistics(self) -> Dict[str, Any]:
        self.logger.info("Step 2: Computing statistics")
        self.statistics = self.stats_engine.analyze(self.result_hurst)
        self.logger.metric("Regime", self.statistics.regime)
        self.logger.metric("R-squared", f"{self.statistics.r_squared:.4f}")
        self.logger.success("Statistics computed")
        return self.stats_engine.summary_dict(self.statistics)

    def _validate(self) -> Dict[str, Any]:
        self.logger.info("Step 3: Validation")

        passed = run_validation(
            self.result_hurst,
            self.statistics,
        )

        if not passed:
            raise RuntimeError("Module 035 full-sample validation failed.")

        if self.rolling_hurst is None:
            raise RuntimeError("Rolling Hurst feature was not calculated.")

        if len(self.rolling_hurst) != len(self.market_df):
            raise RuntimeError("Rolling Hurst row count mismatch.")

        finite = self.rolling_hurst[
            np.isfinite(self.rolling_hurst)
        ]

        if len(finite) < 200:
            raise RuntimeError("Insufficient rolling Hurst observations.")

        if np.unique(np.round(finite, 10)).size < 2:
            raise RuntimeError("Rolling Hurst feature is constant.")

        self.logger.success(
            "Full-sample and rolling Hurst validation passed"
        )

        return {
            "validation_passed": True,
            "rolling_feature_valid": True,
            "rolling_valid_rows": len(finite),
        }

    def _export_results(self) -> Dict[str, Any]:
        self.logger.info("Step 4: Export")

        self.result.artifacts["hurst_statistics"] = (
            self.stats_engine.summary_dataframe(self.statistics)
        )

        hurst_df = pd.DataFrame({
            "scales": self.result_hurst.scales,
            "rs_values": self.result_hurst.rs_values,
        })
        self.result.artifacts["hurst_results"] = hurst_df

        report = HurstReport.generate(
            self.statistics,
            metadata=self.METADATA,
            dataset_name=self.result.dataset_identifier,
        )
        self.result.artifacts["hurst_report"] = report

        # ----------------------------------------------------------
        # Export feature DataFrame for downstream modules
        # ----------------------------------------------------------
        feature_df = self.market_df[["timestamp"]].copy().reset_index(drop=True)

        # Preserve the original full-sample feature.
        feature_df["hurst_exponent"] = (
            self.result_hurst.hurst_exponent
        )

        # New causal rolling feature for Module 103.
        feature_df["hurst_exponent_rolling_60d"] = (
            self.rolling_hurst
        )

        if len(feature_df) != len(self.market_df):
            raise RuntimeError("Module 035 export row count mismatch.")

        if not np.isfinite(
            feature_df["hurst_exponent_rolling_60d"].dropna()
        ).all():
            raise RuntimeError("Invalid rolling Hurst export values.")

        self.result.artifacts["features"] = feature_df

        self.logger.metric("Artifacts", str(len(self.result.artifacts)))
        self.logger.success("Export complete")
        return {
            "export_status": "complete",
            "artifact_count": len(self.result.artifacts),
        }

    def compute(self) -> Dict[str, Any]:
        self.logger.section("Hurst Exponent Pipeline")
        results = self.execute_pipeline()
        self.result.status = "completed"
        self.logger.success("Module 035 complete")
        return results

    def run(self, filepath: str, output_dir=None):
        return super().run(filepath, output_dir)

    def run_dataframe(self, df, dataset_id="in_memory", output_dir=None):
        return super().run_dataframe(df, dataset_id, output_dir)
